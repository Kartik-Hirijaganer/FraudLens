"""Summary: Offline fixed-budget comparison of four frozen risk-blend policies.

Key classes:
- MetricRow: aggregate ranking metrics with exact-budget tie accounting.
- RuleReplay: bounded canonical-rule replay, verified against RuleRegistry.

Key functions:
- main: reconstruct evaluation features, replay rules, select on calibration, evaluate holdout.

Notes:
- No network, database, provider, training, or deployment operations.
- Raw synthetic account keys remain in ignored local Parquet only.
"""

def log(message):
    print(f"[{time.monotonic() - START:.1f}s] {message}", flush=True)


def sha(path):
    h = hashlib.sha256()
    with Path(path).open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def sqltext(value):
    return "'" + str(value).replace("'", "''") + "'"


def connection(work):
    c = duckdb.connect()
    c.execute("SET memory_limit='4GB'")
    c.execute("SET threads=4")
    c.execute("SET preserve_insertion_order=false")
    c.execute(f"SET temp_directory={sqltext(work / 'spill')}")
    return c


class MetricRow(BaseModel):
    name: str = Field(..., description="Policy or baseline name.")
    weight: float | None = Field(None, description="Model share for blended policies.")
    rows: int = Field(..., description="Evaluation transactions.")
    positives: int = Field(..., description="Laundering-labelled transactions.")
    alerts: int = Field(..., description="Exact top-ceil-budget count.")
    true_positives: int = Field(..., description="Labelled positives among alerts.")
    false_positives: int = Field(..., description="Labelled negatives among alerts.")
    recall: float = Field(..., description="Share of positives detected.")
    precision: float = Field(..., description="Share of alerts labelled positive.")
    cutoff: float = Field(..., description="Observed score at the alert-budget boundary.")
    boundary_ties: int = Field(..., description="Rows tied at the score boundary.")
    selected_boundary_ties: int = Field(..., description="Slots filled from the boundary tie.")
    tie_expected_recall: float = Field(..., description="Recall averaged over uniform boundary ties.")


class RuleReplay:
    def __init__(self):
        self.defs = {d.code: d for d in DEFAULT_RULE_DEFINITIONS}
        self.codes = tuple(self.defs)
        self.events = deque()
        self.registry = RuleRegistry()
        self.parity = 0
        self.seen_masks = set()
        self.hit_counts = dict.fromkeys(self.codes, 0)
        self.total_weight = sum(d.weight for d in self.defs.values())
        self.lookup = {}
        for mask in range(1 << len(self.codes)):
            fired = sum((self.defs[code].weight for i, code in enumerate(self.codes)
                         if mask & (1 << i)), Decimal(0))
            self.lookup[mask] = float((fired / self.total_weight).quantize(Decimal('0.0001')))
        self.history_hours = max(int(d.params.get('windowHours', 0)) for d in self.defs.values())
        self.history_max = 100
        self.params = {code: d.params for code, d in self.defs.items()}
        self.ranges = {}
        for code in ('structuring', 'threshold_evasion'):
            param = self.params[code]
            threshold = Decimal(str(param['threshold']))
            self.ranges[code] = (threshold * (1 - Decimal(str(param['marginPct']))), threshold)
        self.ratio = Decimal(str(self.params['rapid_movement']['minRatio']))

    def trim(self, epoch):
        cutoff = epoch - self.history_hours * 3600
        while self.events and self.events[0][0] < cutoff:
            self.events.popleft()

    def append(self, epoch, amount, inbound, country, channel):
        self.events.append((epoch, amount, inbound, country, channel))
        while len(self.events) > self.history_max:
            self.events.popleft()

    def mask(self, epoch, amount, country):
        p = self.params
        lower_s, upper_s = self.ranges['structuring']
        lower_e, upper_e = self.ranges['threshold_evasion']
        hits = {}
        hits['threshold_evasion'] = lower_e <= amount < upper_e
        hits['structuring'] = (int(lower_s <= amount < upper_s) + sum(
            lower_s <= e[1] < upper_s for e in self.events
            if e[0] >= epoch - p['structuring']['windowHours'] * 3600
        )) >= p['structuring']['minCount']
        hits['velocity'] = (1 + sum(e[0] >= epoch - p['velocity']['windowHours'] * 3600
                                  for e in self.events)) > p['velocity']['maxCount']
        hits['round_amount'] = (amount >= p['round_amount']['multipleOf'] and
                               amount % p['round_amount']['multipleOf'] == 0)
        hits['high_risk_geography'] = country in p['high_risk_geography']['countries']
        floor = amount * self.ratio
        hits['rapid_movement'] = any(e[2] and e[1] >= floor and
            e[0] >= epoch - p['rapid_movement']['windowHours'] * 3600 for e in self.events)
        return sum(1 << i for i, code in enumerate(self.codes) if hits[code])

    @staticmethod
    def transaction(epoch, amount, inbound, country, channel):
        return RuleTransaction(amount=amount, currency='XXX', country=country, channel=channel,
            occurred_at=datetime.fromtimestamp(epoch, timezone.utc),
            direction=TransactionDirection.INBOUND if inbound else TransactionDirection.OUTBOUND)

    def score(self, row_id, epoch, amount, country, channel):
        mask = self.mask(epoch, amount, country)
        if mask not in self.seen_masks or row_id % 100003 == 0:
            context = RuleContext(transaction=self.transaction(epoch, amount, False, country, channel),
                history=tuple(self.transaction(*event) for event in self.events))
            expected = self.registry.evaluate(DEFAULT_RULE_DEFINITIONS, context)
            assert not expected.errored_rules
            assert float(expected.subscore) == self.lookup[mask], (row_id, mask, expected)
            assert {h.code for h in expected.hits} == {
                code for i, code in enumerate(self.codes) if mask & (1 << i)}
            self.parity += 1
        self.seen_masks.add(mask)
        for i, code in enumerate(self.codes):
            self.hit_counts[code] += bool(mask & (1 << i))
        return self.lookup[mask]


def prepare(source, config, reference, work):
    dataset = config.dataset(source)
    end_train, end_tuning, end_calibration = reference['folds']['boundary_epochs']
    feature_file = work / 'features.parquet'
    marker = work / 'features-complete.json'
    if marker.exists():
        return json.loads(marker.read_text())
    log(f'{source}: verify and ingest full pinned CSV')
    local_config = config.model_copy(update={'paths': config.paths.model_copy(update={
        'work_dir': str(work.relative_to(ROOT) / 'input')})})
    if reconciliation_path(local_config, dataset, ROOT).exists():
        reconciliation = load_reconciliation(local_config, dataset, ROOT)
        assert sha(ROOT / config.paths.data_dir / dataset.file) == dataset.sha256
    else:
        reconciliation = ingest_dataset(local_config, dataset, ROOT)
    assert reconciliation.input_sha256 == dataset.sha256
    scan = ingested_scan(local_config, dataset, ROOT)
    origin, destination = work / 'origin.parquet', work / 'destination.parquet'
    rules_file = work / 'rules.parquet'
    eval_file = work / 'evaluation.parquet'
    rules_rows = []
    rules_writer = pq.ParquetWriter(rules_file, pa.schema([
        ('row_id', pa.int64()), ('rules_score', pa.float64())]), compression='zstd')
    writers = {1: WindowWriter(origin, BATCH), 2: WindowWriter(destination, BATCH)}
    rules = RuleReplay()
    all_parity = 0
    all_hits = dict.fromkeys(rules.codes, 0)
    masks = set()
    active_account = None
    features = WindowState(config.features.history_max, config.features.window_hours * 3600)
    base_timeline = _timeline_sql(scan).replace('SELECT *,',
        'SELECT origin_account, dest_account, cohort_key, row_id, amount,', 1)
    marker_count = 0
    rule_count = 0
    log(f'{source}: replay bounded histories for calibration + holdout')
    def timeline_rows():
        with connection(work) as c:
            for partition in range(16):
                source_sql = f'read_parquet({sqltext(scan)}, hive_partitioning=false)'
                narrowed = (source_sql + f' WHERE cohort_key >= {end_tuning - 168 * 3600}'
                    + f' AND (hash(origin_account) % 16 = {partition}'
                    + f' OR hash(dest_account) % 16 = {partition})')
                timeline = base_timeline.replace('FROM ' + source_sql, 'FROM ' + narrowed, 1)
                timeline = f'''SELECT * FROM ({timeline})
                    WHERE role = 0 OR occurred_epoch > {end_tuning}'''
                query = f'SELECT * FROM ({timeline}) WHERE hash(account) % 16 = {partition}'
                reader = c.execute(query).to_arrow_reader(batch_size=BATCH)
                for batch in reader:
                    columns = batch.to_pydict()
                    yield from zip(*(columns[k] for k in ('account', 'occurred_epoch', 'row_id', 'role',
                        'amount', 'inbound', 'is_round', 'country_token', 'channel_token')), strict=True)
                log(f'{source}: completed account partition {partition + 1}/16')
    for row in timeline_rows():
                account, epoch, row_id, role, amount, inbound, is_round, country, channel = row
                epoch, row_id, role = int(epoch), int(row_id), int(role)
                if account != active_account:
                    all_parity += rules.parity
                    for code in all_hits:
                        all_hits[code] += rules.hit_counts[code]
                    masks |= rules.seen_masks
                    rules.events.clear()
                    rules.parity = 0
                    rules.hit_counts = dict.fromkeys(rules.codes, 0)
                    rules.seen_masks = masks.copy()
                    features = WindowState(config.features.history_max, config.features.window_hours * 3600)
                    active_account = account
                features.trim(epoch)
                rules.trim(epoch)
                amount_decimal = Decimal(str(amount)).quantize(Decimal('0.01'))
                if role == 0:
                    features.append(_Event(epoch, float(amount), float(inbound), float(is_round),
                                           str(country), str(channel)))
                    rules.append(epoch, amount_decimal, bool(inbound), str(country), str(channel))
                    continue
                writers[role].append(features.snapshot(row_id, epoch, str(country), str(channel)))
                marker_count += 1
                if role == 1:
                    score = rules.score(row_id, epoch, amount_decimal, str(country), str(channel))
                    rules_rows.append({'row_id': row_id, 'rules_score': score})
                    rule_count += 1
                    if len(rules_rows) >= BATCH:
                        rules_writer.write_table(pa.Table.from_pylist(rules_rows, schema=rules_writer.schema))
                        rules_rows.clear()
                if marker_count % 2000000 == 0:
                    log(f'{source}: {marker_count:,} origin/destination markers replayed')
    for writer in writers.values():
        writer.close()
    if rules_rows:
        rules_writer.write_table(pa.Table.from_pylist(rules_rows, schema=rules_writer.schema))
    rules_writer.close()
    all_parity += rules.parity
    for code in all_hits:
        all_hits[code] += rules.hit_counts[code]
    with connection(work) as c:
        log(f'{source}: assemble canonical 19-feature matrix')
        c.execute(f'''COPY (SELECT * FROM read_parquet({sqltext(scan)}, hive_partitioning=false)
            WHERE cohort_key > {end_tuning}) TO {sqltext(eval_file)} (FORMAT PARQUET, COMPRESSION ZSTD)''')
        query = _feature_sql(str(eval_file), origin, destination)
        c.execute(f'''COPY (SELECT f.*, r.rules_score,
            CASE WHEN f.cohort_key <= {end_calibration} THEN 'calibration' ELSE 'holdout' END AS fold
            FROM ({query}) f JOIN read_parquet({sqltext(rules_file)}) r USING(row_id)
            ORDER BY f.cohort_key, f.row_id) TO {sqltext(feature_file)}
            (FORMAT PARQUET, COMPRESSION ZSTD, ROW_GROUP_SIZE 65536)''')
        counts = c.execute(f'''SELECT fold, count(*), sum(label) FROM read_parquet({sqltext(feature_file)})
            GROUP BY fold''').fetchall()
        for fold, count, positives in counts:
            expected = reference['folds']['counts'][fold]
            assert (count, positives) == (expected['rows'], expected['positives'])
    result = {'rules_parity_rows': all_parity, 'rules_hit_counts': all_hits,
        'rule_replay_rows': rule_count, 'feature_sha256': sha(feature_file),
        'source_sha256': dataset.sha256, 'boundary_epochs': reference['folds']['boundary_epochs']}
    marker.write_text(json.dumps(result, indent=2) + '\n')
    return result


def predict(work, artifact):
    path = work / 'predictions.npz'
    if path.exists():
        return
    rows = {'row_id': [], 'label': [], 'rules_score': [], 'probability': [], 'holdout': []}
    artifact.booster.set_param({'nthread': 4})
    for batch in pq.ParquetFile(work / 'features.parquet').iter_batches(batch_size=BATCH):
        frame = batch.to_pydict()
        matrix = np.column_stack([frame[name] for name in FEATURE_NAMES]).astype(np.float32)
        margin = artifact.booster.inplace_predict(matrix, predict_type='margin')
        probability = artifact.calibration.apply(margin)
        for key in ('row_id', 'label', 'rules_score'):
            rows[key].append(np.asarray(frame[key]))
        rows['probability'].append(probability)
        rows['holdout'].append(np.asarray(frame['fold']) == 'holdout')
    np.savez_compressed(path, **{key: np.concatenate(values) for key, values in rows.items()})
    log('frozen-model probabilities cached')


def normalized(probabilities, policy, thresholds):
    if thresholds is None:
        return probabilities
    xs = np.array([0, thresholds.medium, thresholds.high, thresholds.critical, 1])
    ys = np.maximum.accumulate(np.clip([0, policy.band_thresholds['medium'] / policy.model_weight,
        policy.band_thresholds['high'] / policy.model_weight,
        policy.band_thresholds['critical'] / policy.model_weight, 1], 0, 1))
    result = np.interp(probabilities, xs, ys)
    for index in np.linspace(0, len(probabilities) - 1, 100, dtype=int):
        assert abs(result[index] - policy.model_risk(float(probabilities[index]), thresholds)) < 1e-12
    return result


def metrics(name, weight, labels, scores, tie_keys, budget):
    count = max(1, math.ceil(budget * len(labels)))
    order = np.lexsort((tie_keys, -scores))
    chosen = order[:count]
    cutoff = float(scores[chosen[-1]])
    above = scores > cutoff
    equal = scores == cutoff
    slots = count - int(above.sum())
    positives = int(labels.sum())
    tp = int(labels[chosen].sum())
    result = MetricRow(name=name, weight=weight, rows=len(labels), positives=positives,
        alerts=count, true_positives=tp, false_positives=count-tp, recall=tp/positives,
        precision=tp/count, cutoff=cutoff, boundary_ties=int(equal.sum()),
        selected_boundary_ties=slots,
        tie_expected_recall=(float(labels[above].sum()) + slots * float(labels[equal].mean())) / positives)
    return result


def evaluate(source, work, artifact, budget, reference):
    data = np.load(work / 'predictions.npz')
    probability, rules, labels = data['probability'], data['rules_score'], data['label']
    rng = np.random.default_rng(SEED)
    tie_keys = rng.random(len(labels))
    reports = {}
    selected = None
    for fold, is_holdout in (('calibration', False), ('holdout', True)):
        mask = data['holdout'] == is_holdout
        p, r, y, t = probability[mask], rules[mask], labels[mask], tie_keys[mask]
        rows = []
        fixed_model = normalized(p, RiskPolicy(), artifact.risk_thresholds)
        for weight in WEIGHTS:
            policy = RiskPolicy(model_weight=weight)
            model = normalized(p, policy, artifact.risk_thresholds)
            scores = weight * model + (1-weight) * r
            rows.append(metrics(f'{round(weight*100)}/{round((1-weight)*100)}', weight, y, scores, t, budget))
        controls = [metrics('raw_model_only', None, y, p, t, budget),
                    metrics('rules_only', None, y, r, t, budget)]
        sensitivity = [metrics(f'{round(w*100)}/{round((1-w)*100)}', w, y,
            w * fixed_model + (1-w) * r, t, budget) for w in WEIGHTS]
        reports[fold] = {'policies': [row.model_dump() for row in rows],
            'controls': [row.model_dump() for row in controls],
            'fixed_70_normalization': [row.model_dump() for row in sensitivity]}
        if not is_holdout:
            winner = max(rows, key=lambda row: (row.tie_expected_recall, row.weight))
            selected = winner.weight
            selection = {'source': source, 'weight': selected,
                'criterion': 'maximum calibration expected recall under uniform boundary-tie selection',
                'tie_break': 'higher model weight', 'calibration': reports[fold]}
            (work / 'selection-before-holdout.json').write_text(json.dumps(selection, indent=2) + '\n')
            log(f'{source}: calibration selects {winner.name}; selection recorded before holdout metrics')
        else:
            published = reference['evaluation']['metrics']['recall_at_budget']
            reproduced = int(y[np.argsort(-p)[:math.ceil(budget*len(y))]].sum()) / int(y.sum())
            assert abs(reproduced-published) < 1e-12, (reproduced, published)
            reports['baseline_reproduced_recall'] = reproduced
    reports['selected_weight'] = selected
    return reports


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--candidate', default='hi-medium')
    args = parser.parse_args()
    config = load_fulldata_config()
    reference_bytes = (ROOT / 'docs/reference/benchmarks/ibm-full-data-training.json').read_bytes()
    report = json.loads(reference_bytes)['manifest']
    reference = next(s for s in report['sources'] if s['candidate'] == args.candidate)
    work = BASE / args.candidate
    work.mkdir(parents=True, exist_ok=True)
    model = ROOT / '.local/fulldata/artifacts' / reference['evaluation']['modelBundle']
    artifact = load_artifact(model)
    assert sha(model / 'model.json') == reference['evaluation']['modelSha256']
    provenance = prepare(args.candidate, config, reference, work)
    predict(work, artifact)
    results = evaluate(args.candidate, work, artifact, config.gates.alert_budget_fraction, reference)
    payload = {'date': '2026-10-26', 'candidate': args.candidate, 'source': reference['source'],
        'model_bundle': artifact.version_label, 'model_sha256': sha(model / 'model.json'),
        'config_sha256': config.config_sha256, 'reference_sha256': hashlib.sha256(reference_bytes).hexdigest(),
        'script_sha256': sha(__file__), 'budget_fraction': config.gates.alert_budget_fraction,
        'weights': WEIGHTS, 'seed': SEED, 'rule_definitions': [d.model_dump(mode='json') for d in DEFAULT_RULE_DEFINITIONS],
        'provenance': provenance, 'results': results,
        'seconds': time.monotonic()-START,
        'versions': {'numpy': np.__version__, 'duckdb': duckdb.__version__}}
    (work / 'results.json').write_text(json.dumps(payload, indent=2) + '\n')
    log(json.dumps({'selected_weight': results['selected_weight'], 'holdout': results['holdout']['policies']}))


if __name__ == '__main__':
    main()
